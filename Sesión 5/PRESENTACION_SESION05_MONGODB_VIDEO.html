// Comandos CRUD de la Sesión
use concesionario_db;

// 1. Crear colección e insertar 5 documentos
db.parque.insertMany([
  { placa: "ABC-101", marca: "Toyota", modelo: 2022 },
  { placa: "ABC-102", marca: "Chevrolet", modelo: 2023 },
  { placa: "ABC-103", marca: "Renault", modelo: 2021 },
  { placa: "ABC-104", marca: "Nissan", modelo: 2024 },
  { placa: "ABC-105", marca: "Mazda", modelo: 2025 }
]);

// 2. Consulta con filtro por placa
db.parque.find({ placa: "ABC-104" });

// 3. Eliminar documento específico
db.parque.deleteOne({ placa: "ABC-103" });
